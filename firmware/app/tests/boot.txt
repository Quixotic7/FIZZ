# FIZZ in fm1-base's host simulator: firmware/build.py --host, then
#   firmware/app/build/host/fizz_host firmware/app/tests/boot.txt firmware/app/build/host
# Boot, the idle screen, a key into the step view, the demo pattern played, the screen read out and the sound's level.
# The first part follows FIZZ's own render_frames script (tools/render_frames.c: PLAY at 500 ms, frames at 0, 1000,
# 2200 and 2500 ms), so its frames compare with render_frames' at the device's sizes (fzdraw: a frame at that ms).
expect fz_up 1
expect fz_mode 0
fzdraw
shot idle-0000.png
fbdump idle-0000.raw
wait 500
press PLAY
wait 100
release PLAY
wait 400
fzdraw
shot play-1000.png
fbdump play-1000.raw
expect fz_playing 1
expect fz_mode 1
wait 1200
fzdraw
shot play-2200.png
fbdump play-2200.raw
wait 300
fzdraw
shot play-2500.png
fbdump play-2500.raw
# the sound of the demo (A02 at power-on): four seconds of it
peakreset
wav demo.wav
wait 4000
wavstop
expect peak_db_min -36
expect fz_voices >0
expect fz_isr >1000
# MASTER is fm1-base's own gain on the sound (APP_GAIN, the pot's square): full, then low
master 4096
wait 100
peakreset
wait 2000
expect peak_db_min -18
master 600
wait 100
peakreset
wait 2000
expect peak_db_max -36
master 2048
# stopped: PLAY again; a white key held shows the step view's held step
tap PLAY
wait 300
expect fz_playing 0
key w2 down
wait 400
fzdraw
shot held.png
key w2 up
wait 200
# HOME, a knob, the LEDs
tap HOME
wait 100
turn K1 3
wait 300
fzdraw
shot after-knob.png
leds
# FIZZ's SysEx: IDENTITY (F0 7D 46 5A 01 01 7F F7: the command and its checksum sum to 0 in 7 bits) and its reply
sysex F0 7D 46 5A 01 01 7F F7
wait 20
expect fz_sx_frames 1
expect fz_sx_out >0
expect fz_frames >100
# the port's counters (7D 46 50: firmware/tools/fizz_xdata.py stats) and their reset
sysex F0 7D 46 50 F7
sysex F0 7D 46 50 7F F7
expect fz_frames 0
# the store (P2.5): the selected track (SEL + B3: track 3) written by itself, and back after a restart
expect fz_store_on 1
press SEL
tapkey b2
release SEL
wait 100
expect fz_sel 2
wait 4000
expect fz_store_dirty 0
expect fz_store_written >0
reboot
wait 600
expect fz_store_on 1
expect fz_sel 2
# a pattern change while playing (SEQ held + white key 1: A01 queued): the main loop's prefetch brings it, the
# interrupt never loads one itself; and a MIDI note heard from the interrupt
tap PLAY
wait 500
press SEQ
tapkey w0
release SEQ
wait 5000
expect fz_pattern 0
expect fz_isr_loads 0
midi 9F 3C 64
wait 30
expect fz_voices >0
midi 8F 3C 00
tap PLAY
wait 300
