# The store against a region another firmware left (P2.5): noise in 0x97000..0xDFFFF, then a restart: FIZZ finds none
# of its sectors, erases the region and starts on it; the sound costs what it did; the wipe (7D 46 50 "WIPE") and safe
# mode (SAVE held at power-on: no store; P2.14, REC is fm1-base's update mode now). The region: 73 sectors, the Bluetooth
# package's store 49 (persist.h)
fzdata random
reboot
wait 600
expect fz_store_wiped_all 1
expect fz_store_on 1
tap PLAY
wait 3000
expect fz_isr_loads 0
tap PLAY
wait 3000
fzstats
# a region of zeros
fzdata zero
reboot
wait 600
expect fz_store_wiped_all 1
expect fz_store_on 1
# the wipe, FIZZ on: everything FIZZ wrote erased, FIZZ again on a blank store
press SEL
tapkey b2
release SEL
wait 4000
expect fz_sel 2
sysex F0 7D 46 50 57 49 50 45 F7
wait 600
expect fz_store_on 1
expect fz_sel 0
expect fz_store_wiped >0
# safe mode: SAVE held at power-on
press SAVE
reboot
wait 600
release SAVE
expect fz_safe 1
expect fz_store_on 0
expect fz_up 1
expect fz_lcd_stale 0
